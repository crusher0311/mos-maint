**Stack:** Node.js + Express · React · SQLite · Protractor REST API · Detect Dog API

## What this is

A local Node server + Chrome extension that surfaces vehicle maintenance context
the moment a call comes in — before the advisor picks up the phone. It connects
Detect Dog's maintenance intelligence to Protractor's customer and vehicle records,
giving advisors a 3-second head start on every call.

This is a Detect Dog feature. It lives in DD's product, ships under DD's brand,
and is sold as part of DD's value proposition. A clean API hook is included so
App Fueled CRM data can be layered in later for Bloodhound/Top Dog bundle customers.

### Protractor API integration

Base URL: `https://integration.protractor.com/IntegrationServices/1.0/`

Auth — compute HMAC-SHA1 on every request:
```javascript
const crypto = require('crypto');

function protractorHeaders(connectionId, apiKey) {
  const key = apiKey.replace(/-/g, '').toLowerCase();
  const msg = connectionId.replace(/-/g, '').toLowerCase();
  const auth = crypto.createHmac('sha1', key).update(msg).digest('base64');
  return {
    'connectionId': connectionId,
    'apiKey': apiKey,
    'authentication': auth,
    'Accept': 'application/json'
  };
}
```

API call sequence for a phone lookup:
1. `GET /Contact/Search/{phone}` → returns contact record with ID
2. `GET /ServiceItem/Search/OwnerID/{contactId}` → returns vehicles
3. `GET /WorkOrder/` → filter by contactId, status=Open (client-side filter if
   endpoint doesn't support query params — flag this for Brandon to confirm)

### content.js — phone number detection

```javascript
const PHONE_RE = /\b(\+?1[-.\s]?)?\(?\d{3}\)?[-.\s]?\d{3}[-.\s]?\d{4}\b/g;

function normalizePhone(raw) {
  return raw.replace(/\D/g, '').slice(-10);
}

// Debounce to avoid duplicate triggers on DOM churn
let lastPhone = null;
let debounceTimer = null;

function checkForIncomingCall() {
  // Broad selector — catches most VoIP web apps
  // TODO: tune these selectors to the specific VoIP app Heart Auto Care uses
  const indicators = document.querySelectorAll([
    '[class*="incoming"]',
    '[class*="ringing"]',
    '[class*="inbound"]',
    '[class*="call-alert"]',
    '[class*="caller"]',
    '[data-call-state="ringing"]',
    '[data-call-state="incoming"]'
  ].join(','));

  indicators.forEach(el => {
    const match = el.textContent.match(PHONE_RE);
    if (!match) return;
    const phone = normalizePhone(match[0]);
    if (phone === lastPhone) return;
    lastPhone = phone;
    clearTimeout(debounceTimer);
    debounceTimer = setTimeout(() => { lastPhone = null; }, 30000);
    chrome.runtime.sendMessage({ type: 'INCOMING_CALL', phone });
  });
}

const observer = new MutationObserver(checkForIncomingCall);
observer.observe(document.body, { childList: true, subtree: true });
```

### background.js

```javascript
chrome.runtime.onMessage.addListener(async (msg) => {
  if (msg.type !== 'INCOMING_CALL') return;

  let data;
  try {
    const res = await fetch('http://localhost:7432/lookup', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ phone: msg.phone })
    });
    data = await res.json();
  } catch (e) {
    // Server not running or unreachable — fail silently
    console.warn('Detect Dog server not reachable:', e.message);
    return;
  }

  // Open caller card as a small popup window
  const params = encodeURIComponent(JSON.stringify(data));
  chrome.windows.create({
    url: chrome.runtime.getURL(`card/card.html?data=${params}`),
    type: 'popup',
    width: 400,
    height: 560,
    top: 40,
    left: 40
  });
});
```

## Module 5 — Windows Setup

**File:** `setup.bat`

Registers the `protractorbridge://` URI scheme and starts the server:

```bat
@echo off
echo Installing Detect Dog Caller ID...

REM Register URI scheme
reg add "HKCU\Software\Classes\protractorbridge" /ve /d "URL:Protractor Bridge Protocol" /f
reg add "HKCU\Software\Classes\protractorbridge" /v "URL Protocol" /d "" /f
reg add "HKCU\Software\Classes\protractorbridge\shell\open\command" /ve /d "\"%~dp0shim\protractorbridge.exe\" \"%%1\"" /f

REM Add to Windows startup
set STARTUP=%APPDATA%\Microsoft\Windows\Start Menu\Programs\Startup
copy "%~dp0start-server.bat" "%STARTUP%\DetectDogCallerID.bat" /y

echo Done. Starting server...
start "" "%~dp0start-server.bat"
echo Open http://localhost:7432 to configure your credentials.
pause
```

**File:** `shim/protractorbridge-shim.js` (Node script compiled to .exe via pkg)

```javascript
const { execSync, spawn } = require('child_process');
const uri = process.argv[2]; // e.g. protractorbridge://contact/GUID

const contactId = uri.replace('protractorbridge://contact/', '');
const protractorExe = 'C:\\Program Files\\Protractor Software\\Protractor.Net Shop Manager\\Protractor.Launcher.ShopManager.exe';

// Check if Protractor is running
try {
  const procs = execSync('tasklist /FI "IMAGENAME eq Protractor.Client.ShopManager.exe"').toString();
  if (procs.includes('Protractor.Client.ShopManager.exe')) {
    // Already running — pass contact ID as arg (StartupNextInstance handles it)
    spawn(protractorExe, [`/contact:${contactId}`], { detached: true });
  } else {
    // Launch fresh with contact arg
    spawn(protractorExe, [`/contact:${contactId}`], { detached: true });
  }
} catch (e) {
  console.error('Could not launch Protractor:', e.message);
}
```
